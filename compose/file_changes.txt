//Production Docker File
# Use a specific version of Node.js for consistency and security
FROM node:20.18.1-alpine

# Set environment to production for optimized dependency installation
ENV NODE_ENV=production

# Set the working directory inside the container
WORKDIR /app

# Copy package files to install dependencies
COPY package.json package-lock.json ./

# Install only production dependencies
RUN npm ci --only=production

# Copy the rest of the application code
COPY . .

# (Optional) Build the application if required
RUN npm run build

# Expose the port your app runs on (if needed)
EXPOSE 3000

# Use a production-ready command to start the application
CMD ["npm", "start"]
